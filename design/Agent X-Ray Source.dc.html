agent-xray
一个把 agent loop 摊开给人看的站点。访客在左边和 agent 对话,右边像 DevTools 一样实时显示这一轮的事件流、工具调用、上下文占用与耗时。它不是 SDK,也不是框架——就是一份跑得起来、读得完的内核参照。
它由什么组成
仓库是一个 monorepo。站点在 apps/web(Next.js App Router),服务在 apps/api(Encore.ts)——agent loop、工具注册表、权限闸都在后者里,一共不到三千行。
·apps/api/agent —— 循环本体、工具表、事件流编码
·runner —— 命令执行的沙箱容器,与 API 分离部署
·deploy / tools —— compose 编排与发版脚本(镜像 tag 就是 git SHA)
·docs / rounds —— 设计决策与每一轮的任务卡
本地跑起来
需要 Node 20、pnpm 9 和一份 .env.local(至少一个模型 API key)。三条命令起全栈:
powershell
copy
pnpm i
./dev.ps1 up      # 起 api + web + runner
./dev.ps1 check   # 类型 / lint / 测试一把过
这一页的源码是怎么来的
你现在看到的每个文件都是线上正在跑的那一版的快照:发版时按 git SHA 打包成只读文件表,和镜像一起送上服务器。所以页头那个 be6c074 既是这堆文件的版本,也是当前进程的版本。
许可
MIT。拿去改、拿去讲课、拿去当自己项目的起点都行;只求别把「看得见的内核」这件事做丢了。
这是本站自己的源码,与线上运行的版本一致;快照随每次发版更新。

// tools.ts —— 工具注册表:schema、权限闸、执行体三件事在这里收口。
// 新增工具只改这一处;agent loop 不认识任何具体工具,只认识这张表。
import { z } from "zod";
import { runShell } from "../runtime/sandbox";
import type { ToolContext, ToolResult } from "./types";
/** 高危模式:命中即拒,不进沙箱。 */
const HIGH_RISK = /rm\s+-rf|sudo|mkfs|dd\s+if=/;
export const tools = {
notes_search: {
label: "教程库检索",
group: "read",
schema: z.object({
query: z.string().min(2).max(300),
limit: z.number().int().min(1).max(20).default(5),
}),
// 只读工具不过权限闸,直接打库。
async run({ query, limit }, ctx: ToolContext): Promise<ToolResult> {
const hits = await ctx.db.search(query, limit);
return { ok: true, text: JSON.stringify({ hits }) };
},
},
bash: {
label: "命令执行", group: "write",
schema: z.object({ cmd: z.string().min(1).max(2000) }),
async run({ cmd }, ctx: ToolContext): Promise<ToolResult> {
// permission-gate:拒绝理由原样回给模型,让它自己改写命令 —— 下面这一行故意留得很长,用来交代长行的处理:它在本行内横向滚动,行号列与页面都不动。
if (HIGH_RISK.test(cmd)) return { ok: false, text: `permission-gate returned deny: 命令 ${cmd} 命中高危模式 ${HIGH_RISK.source},已拒绝执行;如确需该操作请改写为只读命令,或由维护者在服务端手工执行。` };
return runShell(cmd, { cwd: ctx.workdir, timeoutMs: 30_000 });
},
},
};
